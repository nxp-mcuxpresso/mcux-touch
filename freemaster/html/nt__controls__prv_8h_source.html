279 static inline int32_t _nt_control_overrun(struct nt_control_data *control)
280 {
281  if ((bool)_nt_control_get_flag(control, (int32_t)NT_CONTROL_NEW_DATA_FLAG))
282  {
283  return (int32_t)NT_FAILURE; /* overrun, data has not been processed */
284  }
285  _nt_control_clear_flag(control, (int32_t)NT_CONTROL_NEW_DATA_FLAG);
286  return (int32_t)NT_SUCCESS;
287 }
